let number = 10;

if(number % 2 === 0){
    console.log(`The number ${number} is an Even no. `);
}else{
    console.log(`This number ${number} is an Odd no. `);
}

// In this if else condition the condition check if a number is divided by 2 and remainder is 0 then execute the first condtion, 
// next If the remainder is not equal to 0 and execute a remainder then execute the 2nd condition.

// Here you see i use % operator for check remainder is 0 or not.. 
// This % operator is returns leftover amount after divide a number by another no.
